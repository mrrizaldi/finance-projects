def oklch_ke_linear(L, C, H):
    a = C * math.cos(math.radians(H))
    b = C * math.sin(math.radians(H))
    l_ = L + 0.3963377774 * a + 0.2158037573 * b
    m_ = L - 0.1055613458 * a - 0.0638541728 * b
    s_ = L - 0.0894841775 * a - 1.2914855480 * b
    l, m, s = l_ ** 3, m_ ** 3, s_ ** 3
    r = 4.0767416621 * l - 3.3077115913 * m + 0.2309699292 * s
    g = -1.2684380046 * l + 2.6097574011 * m - 0.3413193965 * s
    bl = -0.0041960863 * l - 0.7034186147 * m + 1.7076147010 * s
    return [min(1.0, max(0.0, x)) for x in (r, g, bl)]


def gamma(c):
    return 12.92 * c if c <= 0.0031308 else 1.055 * c ** (1 / 2.4) - 0.055


def ke_hex(lin):
    return "#%02X%02X%02X" % tuple(round(gamma(c) * 255) for c in lin)


def luminansi(hx):
    out = []
    for i in (1, 3, 5):
        c = int(hx[i:i + 2], 16) / 255
        out.append(c / 12.92 if c <= 0.04045 else ((c + 0.055) / 1.055) ** 2.4)
    r, g, b = out
    return 0.2126 * r + 0.7152 * g + 0.0722 * b


def rasio(a, b):
    la, lb = luminansi(a), luminansi(b)
    return (max(la, lb) + 0.05) / (min(la, lb) + 0.05)

const BASE = 'http://localhost:3000';
const VIEWPORTS = { desktop: { width: 1280, height: 800 }, mobile: { width: 390, height: 844 } };

async function shot(page, name) {
  const file = path.join(__dirname, `${name}.png`);
  await page.waitForLoadState('load');
  await page.waitForTimeout(1500);
  await page.screenshot({ path: file, fullPage: true });
  console.log(file);
}

(async () => {
  const browser = await chromium.launch();
  for (const theme of ['light', 'dark']) {
    for (const [vp, viewport] of Object.entries(VIEWPORTS)) {
      const ctx = await browser.newContext({ viewport, colorScheme: theme });
      await ctx.addInitScript((t) => localStorage.setItem('theme', t), theme);
      const page = await ctx.newPage();
      await page.goto(`${BASE}/login`);
      await shot(page, `login-${theme}-${vp}`);
      if (EMAIL && PASSWORD) {
        await page.fill('#email', EMAIL);
        await page.fill('#password', PASSWORD);
        await page.click('form button[type="submit"]');
        await page.waitForURL(`${BASE}/`);
        for (const [name, url] of Object.entries(PAGES)) {
          await page.goto(`${BASE}${url}`);
          await shot(page, `${name}-${theme}-${vp}`);
        }
      } else {
        console.log('E2E_EMAIL/E2E_PASSWORD kosong, halaman dalam dilewati');
      }
      await ctx.close();
    }
  }
  await browser.close();
})();
